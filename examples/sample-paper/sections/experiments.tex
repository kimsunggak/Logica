\section{Experiments}
\label{sec:experiments}

\subsection{Dataset}
We evaluate on a video pose benchmark with annotated occlusion labels.

\subsection{Metrics}
We report the mean per-joint position error (MPJPE) on all joints and on
occluded joints only.

\subsection{Main Results}
Table~\ref{tab:main} compares \method{} with a per-frame baseline and a
temporal baseline \citep{pavllo2019videopose}.

  \centering

    \toprule
    Method & All joints & Occluded \\
    \midrule

    Temporal & 48.7 & 69.0 \\
    \method{} (ours) & \textbf{46.9} & \textbf{61.2} \\
    \bottomrule
  \end{tabular}
  \caption{MPJPE (lower is better).}

\end{table}
